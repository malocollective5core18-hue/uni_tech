academics/models.pyreplace contents
from django.db import models
from django.contrib.auth.models import User

class Teacher(models.Model):
    user = models.OneToOneField(User, on_delete=models.CASCADE)
    department = models.CharField(max_length=100)

    def __str__(self):
        return self.user.get_full_name() or self.user.username


class Student(models.Model):
    user = models.OneToOneField(User, on_delete=models.CASCADE)
    admission_number = models.CharField(max_length=20, unique=True)
    grade_level = models.CharField(max_length=20)

    def __str__(self):
        return self.user.get_full_name() or self.user.username


class Course(models.Model):
    title = models.CharField(max_length=150)
    slug = models.SlugField(unique=True)
    description = models.TextField(blank=True)
    teacher = models.ForeignKey(Teacher, related_name='courses', on_delete=models.SET_NULL, null=True)
    capacity = models.PositiveIntegerField(default=30)

    def __str__(self):
        return self.title

    def seats_taken(self):
        return self.enrollments.count()
Why OneToOneField(User, ...) on both Teacher and StudentRather than reinventing login/password/email, we attach a small "profile" of extra fields to Django's already-battle-tested User model — the same pattern applied twice here for two different roles.
Why on_delete=SET_NULL on Course.teacher, not CASCADEDeleting a teacher's account shouldn't delete every course they ever taught — the course should survive with an unset teacher, ready to be reassigned. (Requires null=True, added above.)
Why seats_taken() calls self.enrollments.count() before Enrollment even existsWriting the method against the related_name already planned for Step 15's Enrollment model — same "write against the interface I know I need" habit as the cart project. It simply won't work until that model exists; that's fine, we're not calling it yet.

enrollment/models.pynew
from django.db import models
from academics.models import Student, Course

class Enrollment(models.Model):
    student = models.ForeignKey(Student, related_name='enrollments', on_delete=models.CASCADE)
    course = models.ForeignKey(Course, related_name='enrollments', on_delete=models.CASCADE)
    enrolled_at = models.DateTimeField(auto_now_add=True)
    attended = models.BooleanField(default=False)

    class Meta:
        unique_together = ('student', 'course')

    def __str__(self):
        return f"{self.student} in {self.course}"

related_name='enrollments' here is exactly the name Course.seats_taken() (Step 4) was already written against — this is the moment that method starts actually working.

unique_together = ('student', 'course') prevents a student from enrolling in the same course twice — a database-level constraint, not just something we hope the view remembers to check.

enrollment/views.pynew
from django.shortcuts import render, redirect, get_object_or_404
from django.contrib.auth import login, logout
from django.contrib.auth.decorators import login_required
from django.contrib.auth.forms import UserCreationForm, AuthenticationForm
from django.views.decorators.http import require_POST
from academics.models import Student, Course
from .models import Enrollment

def register(request):
    if request.method == 'POST':
        form = UserCreationForm(request.POST)
        if form.is_valid():
            user = form.save()
            Student.objects.create(
                user=user,
                admission_number=f"STU{user.id:05d}",
                grade_level='Unassigned',
            )
            login(request, user)
            return redirect('enrollment:my_dashboard')
    else:
        form = UserCreationForm()
    return render(request, 'enrollment/register.html', {'form': form})

def login_view(request):
    if request.method == 'POST':
        form = AuthenticationForm(request, data=request.POST)
        if form.is_valid():
            login(request, form.get_user())
            return redirect('enrollment:my_dashboard')
    else:
        form = AuthenticationForm()
    return render(request, 'enrollment/login.html', {'form': form})

@require_POST
def logout_view(request):
    logout(request)
    return redirect('academics:course_list')

@login_required
def enroll(request, course_id):
    course = get_object_or_404(Course, id=course_id)
    student = get_object_or_404(Student, user=request.user)
    Enrollment.objects.get_or_create(student=student, course=course)
    return redirect('academics:course_detail', slug=course.slug)

@login_required
def drop(request, course_id):
    student = get_object_or_404(Student, user=request.user)
    Enrollment.objects.filter(student=student, course_id=course_id).delete()
    return redirect('enrollment:my_dashboard')

@login_required
def my_dashboard(request):
    student = Student.objects.filter(user=request.user).first()
    if student:
        enrollments = Enrollment.objects.filter(student=student)
        return render(request, 'enrollment/student_dashboard.html', {'enrollments': enrollments})
    # Not a student — check if they're a teacher instead
    from academics.models import Teacher
    teacher = get_object_or_404(Teacher, user=request.user)
    courses = Course.objects.filter(teacher=teacher)
    return render(request, 'enrollment/teacher_dashboard.html', {'courses': courses})
Why get_or_create in enroll, not plain createMatches the unique_together constraint from Step 15 — trying to create() a duplicate enrollment would raise IntegrityError; get_or_create handles "already enrolled" gracefully instead of crashing.
Why my_dashboard branches on Student-vs-TeacherThis app's registration flow only creates Students — but the earlier admin-seeded Ms. Alvarez (Step 7) is a Teacher, not a Student. One dashboard URL routing to two different templates keeps the URL structure simple (LOGIN_REDIRECT_URL from Step 11 only needs to point at one place) while still serving each role correctly.

@require_POST
@login_required
def mark_attendance(request, enrollment_id):
    enrollment = get_object_or_404(Enrollment, id=enrollment_id, course__teacher__user=request.user)
    enrollment.attended = True
    enrollment.save()
    return redirect('enrollment:my_dashboard')
This lookup is doing real security workcourse__teacher__user=request.user guarantees a teacher can only mark attendance for their own course's enrollments — a teacher can't hit this URL with an arbitrary enrollment_id and mark a student in someone else's class.

def register(request):

    if request.method == 'POST' and form.is_valid():

    return render(request, 'enrollment/auth.html', {'form': form, 'mode': 'Create your student account'})

def login_view(request):
    form = AuthenticationForm(request, data=request.POST or None)
    if request.method == 'POST' and form.is_valid(): login(request, form.get_user()); return redirect('enrollment:dashboard')

@login_required
@require_POST
def enroll(request, course_id):
    student = get_object_or_404(Student, user=request.user); course = get_object_or_404(Course, id=course_id)
    if course.seats_taken() < course.capacity: Enrollment.objects.get_or_create(student=student, course=course)
    return redirect('enrollment:dashboard')

@login_required
@require_POST
def drop(request, course_id):
    Enrollment.objects.filter(student__user=request.user, course_id=course_id).delete(); return redirect('enrollment:dashboard')

def dashboard(request):
    student = Student.objects.filter(user=request.user).first()
    if student: return render(request, 'enrollment/student_dashboard.html', {'enrollments': student.enrollments.select_related('course', 'course__teacher')})
    teacher = get_object_or_404(Teacher, user=request.user)
    return render(request, 'enrollment/teacher_dashboard.html', {'courses': teacher.courses.prefetch_related('enrollments__student__user')})

@require_POST

The pattern worth remembering

Same loop as the e-commerce build.

Model→ Migrate→ Admin (seed data)→ View→ URL→ Template

Every time a template linked to something that didn't exist yet (enrollment:enroll, enrollment:drop, enrollment:mark_attendance), that link became the very next thing built. The only genuinely new idea in this project — the security check inside mark_attendance (course__teacher__user=request.user) — is worth internalizing beyond this one app: any time a logged-in user can act on a database row by ID, ask "what stops them from passing someone else's ID?" before calling the view finished.
